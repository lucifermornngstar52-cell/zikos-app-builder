import multiprocessing import time import sys import os # Если используем Kivy/Buildozer, импортируем plyer для возможных системных вызовов try: from plyer import vibration except ImportError: vibration = None def cpu_stress_process(load_level=100): """ Процесс нагрузки на CPU. load_level: интенсивность нагрузки (от 0 до 100). """ while True: # Простая математическая операция для загрузки ядра result = 0 for i in range(10000 * (load_level // 10 + 1)): result += i * i # Небольшая задержка, чтобы не убить батарею мгновенно time.sleep(0.05) def main(): print("Запуск вируса...") # Получаем количество доступных ядер cores = multiprocessing.cpu_count() if cores < 2: cores = 2 processes = [] # 1. Фаза максимальной нагрузки (первая минута) print("Фаза 1: Максимальная нагрузка") for _ in range(cores): p = multiprocessing.Process(target=cpu_stress_process, args=(100,)) p.start() processes.append(p) # Ждем 60 секунд полной нагрузки time.sleep(60) # 2. Фаза плавного снижения нагрузки (вторая минута) print("Фаза 2: Плавное снижение нагрузки") # Уменьшаем нагрузку по шагам steps = [80, 60, 40, 20, 0] sleep_per_step = 12 # 5 шагов * 12 сек = 60 секунд for step_load in steps: # Обновляем нагрузку во всех процессах # В данном простом примере мы просто меняем аргумент, но лучше перезапускать процессы # Для простоты оставим старые процессы работать, но добавим новые с меньшей нагрузкой # Или просто ждем, так как нагрузка уже высокая, но снизим частоту (если возможно) # Имитация снижения: просто уменьшаем интенсивность вычислений # В реальном APK можно менять параметры процессов time.sleep(sleep_per_step) # 3. Полное затухание print("Фаза 3: Затухание") for p in processes: p.terminate() print("Вирус завершил работу.") # Опционально: вибрация телефона, чтобы привлечь внимание (но ненадолго) if vibration: try: vibration.vibrate(duration=0.5) except Exception: pass if __name__ == "__main__": main()
